// verify_features.mjs —— 2026-08-08 新增功能端到端验收
// 覆盖：考研倒计时预填 / 顶部考试切换 / 深呼吸默认展开 / 自测模块 / 随机真题考试（限时+锁App+判分+解析+知识点+薄弱点）/ 估分写回
import puppeteer from 'puppeteer-core'
import { mkdirSync } from 'node:fs'
import path from 'node:path'

const URL = 'http://127.0.0.1:4173'
const EDGE = 'C:\\Program Files (x86)\\Microsoft\\Edge\\Application\\msedge.exe'
const EMAIL = process.env.PV_EMAIL || 'u-pv-1786159490551@example.com'
const PASS = 'Test123456'
const OUT = 'C:/Users/Administrator/AppData/Roaming/haolo_desktop/thread-groups/default/app/outputs/study-buddy-preview'
mkdirSync(OUT, { recursive: true })
const sleep = (ms) => new Promise((r) => setTimeout(r, ms))

const browser = await puppeteer.launch({ executablePath: EDGE, headless: 'new', args: ['--no-sandbox', '--disable-gpu'] })
const page = await browser.newPage()
await page.setViewport({ width: 390, height: 844, deviceScaleFactor: 2 })
const results = []
const errs = []
page.on('pageerror', (e) => errs.push('pageerror: ' + e.message))
page.on('console', (m) => { if (m.type() === 'error') errs.push('console.error: ' + m.text()) })
page.on('dialog', async (d) => { console.log('[dialog] ' + d.message()); await d.accept() })

const ok = (name, pass, extra = '') => {
  results.push({ name, pass: !!pass })
  console.log((pass ? 'PASS' : 'FAIL') + ' ' + name + (extra ? ' :: ' + extra : ''))
}
const bodyText = () => page.evaluate(() => document.body.innerText)
const clickByText = async (txt, tag = 'button') => {
  const okc = await page.evaluate(({ txt, tag }) => {
    const el = [...document.querySelectorAll(tag)].find((e) => (e.textContent || '').includes(txt))
    if (!el) return false
    el.click(); return true
  }, { txt, tag })
  if (!okc) throw new Error('element not found: ' + txt)
}
const typeByPlaceholder = async (txt, val) => {
  const h = await page.evaluateHandle((txt) => [...document.querySelectorAll('input')].find((e) => (e.placeholder || '').includes(txt)) || null, txt)
  const el = h.asElement()
  if (!el) throw new Error('input not found: ' + txt)
  await el.click(); await el.type(val)
}
const tapTab = async (label) => {
  await page.evaluate((label) => {
    const el = [...document.querySelectorAll('.tabs button')].find((e) => (e.textContent || '').trim() === label)
    el && el.click()
  }, label)
}

// 登录
await page.goto(URL, { waitUntil: 'networkidle2', timeout: 30000 })
await sleep(1000)
await typeByPlaceholder('you@example.com', EMAIL)
await typeByPlaceholder('至少 6 位', PASS)
await clickByText('进入')
await page.waitForFunction(() => document.body.innerText.includes('今天还没打卡'), { timeout: 30000 })
await sleep(2000)

// 1) 倒计时预填 + 顶部考试切换
let t = await bodyText()
ok('考研倒计时已预填（2026-12-19）', t.includes('考研初试') && t.includes('距离'), '')
t = await bodyText()
ok('顶部考试切换按钮', t.includes('当前考试：考研'), '')
await page.screenshot({ path: path.join(OUT, '11-today-cd.png') })

// 2) 深呼吸默认展开（专注页）
await tapTab('专注')
await sleep(1200)
t = await bodyText()
ok('深呼吸默认展开（可见开始按钮）', t.includes('深呼吸') && t.includes('开始 60 秒呼吸'), '')
await page.screenshot({ path: path.join(OUT, '12-breathing.png') })

// 3) 自测模块 + 随机考试
await tapTab('自测')
await sleep(2200)
t = await bodyText()
ok('自测模块（正式考试 + 真题库）', t.includes('正式考试') && t.includes('真题库') && t.includes('随机组卷'), '')
await page.screenshot({ path: path.join(OUT, '13-selftest.png') })

await clickByText('开始随机考试')
await sleep(1500)
t = await bodyText()
ok('考试开始（限时 + 逐题）', t.includes('第 1/') && (t.includes('180:00') || t.includes('179:5')), '')
ok('考试中全屏锁 App 遮罩', await page.evaluate(() => !!document.querySelector('.quiz-overlay')), '')
await page.screenshot({ path: path.join(OUT, '14-exam.png') })

// 4) 考试中切换 Tab 被拦截
await tapTab('备考')
await sleep(600)
const stillExam = await page.evaluate(() => !!document.querySelector('.quiz-overlay'))
ok('考试中切 Tab 被拦截', stillExam, '')

// 5) 答前 6 题（全选 A，必错若干）后交卷
for (let i = 0; i < 6; i++) {
  await page.evaluate(() => {
    const opt = document.querySelector('.quiz-opt')
    opt && opt.click()
    const next = [...document.querySelectorAll('.quiz-nav button')].find((b) => (b.textContent || '').includes('下一题'))
    next && next.click()
  })
  await sleep(120)
}
await clickByText('交卷')
await sleep(2500)
t = await bodyText()
ok('交卷自动判分（总分）', t.includes('考试结果') && t.includes('选择题'), '')
ok('各科表现（薄弱点）', t.includes('各科表现'), '')
ok('错题回顾 + 解析 + 知识点', t.includes('错题回顾') && t.includes('解析：') && t.includes('知识点：'), '')
await page.screenshot({ path: path.join(OUT, '15-exam-result.png') })

// 6) 回到进度页：估分应已可用
await clickByText('完成')
await sleep(800)
await tapTab('进度')
await page.waitForFunction(() => document.body.innerText.includes('连续天数'), { timeout: 20000 })
await sleep(2500)
t = await bodyText()
ok('离岸线估分可用（出现数字而非 —）', t.includes('当前估分') && /\d+/.test((t.match(/当前估分\s*\d+/) || [''])[0]), '')
await page.screenshot({ path: path.join(OUT, '16-progress-est.png') })

console.log(`\n结果: ${results.filter((r) => r.pass).length}/${results.length} 通过`)
if (errs.length) { console.log('页面错误:'); errs.slice(0, 8).forEach((e) => console.log('  ' + e)) } else console.log('无页面错误')
await browser.close()
process.exit(results.some((r) => !r.pass) ? 1 : 0)
