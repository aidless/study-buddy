# -*- coding: utf-8 -*-
"""Minimal trusted self-evolving loop (Plan C orchestrator).

Usage:
  python -m agent.cycle --phase all --smoke
  python -m agent.cycle --phase all
"""

from __future__ import annotations

import argparse
import json
import pathlib
import re
import sys
from typing import Any

from .config import load_control_plane
from .gate import decide_promotion, resolved
from .ledger import Ledger
from .llm_client import ChatClient, LLMError
from .runner import load_prompt, run_arm
from .skills import curate_candidate, placebo_block, retrieve_skills_block, SkillLibrary
from .tools import RepairEnv

ROOT = pathlib.Path(__file__).resolve().parent.parent


def verify_task_bank(cfg: Any) -> list[str]:
    errors: list[str] = []
    for task in cfg.tasks:
        with RepairEnv(task, timeout_s=cfg.budgets["test_timeout_s"]) as env:
            pub = env.run_public_tests()
            hid = env.evaluate_buggy_hidden()
        if not pub["passed"]:
            errors.append(f"{task['id']}: buggy source fails its public test")
        if hid["passed"]:
            errors.append(f"{task['id']}: buggy source passes its hidden test (not discriminating)")
    return errors


def parse_json_object(text: str) -> dict[str, Any] | None:
    cleaned = re.sub(r"```(?:json)?\s*|\s*```", "", text)
    start, end = cleaned.find("{"), cleaned.rfind("}")
    if start == -1 or end <= start:
        return None
    try:
        obj = json.loads(cleaned[start : end + 1])
        return obj if isinstance(obj, dict) else None
    except json.JSONDecodeError:
        return None


def _ids(value: str, default: list[str]) -> list[str]:
    return [s.strip() for s in value.split(",") if s.strip()] if value else default


def _save(state_dir: pathlib.Path, name: str, obj: Any) -> None:
    (state_dir / name).write_text(
        json.dumps(obj, ensure_ascii=False, indent=2), encoding="utf-8"
    )


def _load(state_dir: pathlib.Path, name: str) -> Any:
    path = state_dir / name
    if not path.exists():
        return None
    return json.loads(path.read_text(encoding="utf-8"))


class Cycle:
    def __init__(self, args: argparse.Namespace):
        self.args = args
        self.cfg = load_control_plane()
        self.state_dir = pathlib.Path(args.state_dir)
        self.state_dir.mkdir(parents=True, exist_ok=True)
        self.ledger = Ledger(self.state_dir / "ledger.jsonl")
        self.client = ChatClient(
            args.endpoint or self.cfg.endpoint["base_url"],
            args.model or self.cfg.endpoint["model"],
            temperature=self.cfg.sampling["temperature"],
            seed=self.cfg.sampling["seed"],
            max_tokens=self.cfg.sampling["max_tokens"],
            timeout_s=self.cfg.budgets["generation_timeout_s"],
            max_retries=self.cfg.budgets["max_retries"],
        )

    def default_splits(self) -> tuple[list[str], list[str], list[str]]:
        acq_all = [t["id"] for t in self.cfg.tasks_by_split("acquisition")]
        val_all = [t["id"] for t in self.cfg.tasks_by_split("validation")]
        sealed_all = [t["id"] for t in self.cfg.tasks_by_split("sealed_final")]
        if self.args.smoke:
            return acq_all[:6], val_all, sealed_all
        return acq_all, val_all, sealed_all

    def start(self) -> None:
        errors = Ledger.verify(self.ledger.path)
        if errors:
            print("LEDGER INVALID:", errors)
            sys.exit(1)
        self.ledger.append(
            "control_plane_frozen",
            {"fingerprint": self.cfg.fingerprint(), "args": vars(self.args)},
        )
        print(f"control plane: {json.dumps(self.cfg.fingerprint(), ensure_ascii=False)}")

    def phase_validate_bank(self) -> None:
        errors = verify_task_bank(self.cfg)
        if errors:
            print("TASK BANK INVALID:")
            for e in errors:
                print(" -", e)
            sys.exit(1)
        self.ledger.append("task_bank_validated", {"tasks": len(self.cfg.tasks)})
        print(f"task bank OK: {len(self.cfg.tasks)} tasks")

    def phase_acquire(self) -> list[dict[str, Any]]:
        acq_ids, _, _ = self.default_splits()
        acq_ids = _ids(self.args.acq_ids, acq_ids)
        tasks = self.cfg.tasks_by_ids(acq_ids)
        print(f"[acquire] {len(tasks)} tasks, policy=base")
        results = run_arm(self.client, tasks, self.cfg, policy="base")
        _save(self.state_dir, "acquire_results.json", results)
        self.ledger.append(
            "acquisition_run",
            {
                "tasks": acq_ids,
                "resolved": [r["task_id"] for r in results if r.get("resolved")],
                "policy": "base",
            },
        )
        print(f"[acquire] resolved {sum(1 for r in results if r.get('resolved'))}/{len(results)}")
        return results

    def phase_diagnose(self, acquire_results: list[dict[str, Any]]) -> list[dict[str, Any]]:
        failures = [r for r in acquire_results if not r.get("resolved")]
        print(f"[diagnose] {len(failures)} failures")
        curated: list[dict[str, Any]] = []
        system = load_prompt("diagnose_system.txt")
        for failure in failures:
            task = self.cfg.task(failure["task_id"])
            user = (
                f"Failed task: {task['id']} (category: {task['category']})\n\n"
                f"Instruction:\n{task['instruction']}\n\n"
                f"Public test:\n```python\n{task['public_test']}\n```\n\n"
                f"Submitted code:\n```python\n{failure.get('final_code') or '(none)'}\n```\n\n"
                f"Public feedback:\n{failure.get('public_feedback') or '(none)'}\n\n"
                "Propose transferable skills. Output only the JSON object."
            )
            try:
                resp = self.client.complete(
                    [
                        {"role": "system", "content": system},
                        {"role": "user", "content": user},
                    ],
                    max_tokens=self.args.diagnose_tokens,
                    temperature=0.0,
                )
                content = resp["message"].get("content", "")
            except LLMError as exc:
                self.ledger.append("diagnose_error", {"task": task["id"], "error": str(exc)})
                continue
            obj = parse_json_object(content)
            candidates = (obj or {}).get("candidate_skills", [])
            for cand in candidates[: self.cfg.curator["max_candidates_per_failure"]]:
                cand["evidence_ref"] = task["id"]
                skill, reason = curate_candidate(cand, self.cfg.by_id, self.cfg.data)
                if skill is None:
                    self.ledger.append(
                        "candidate_rejected_curation",
                        {"task": task["id"], "reason": reason},
                    )
                    print(f"  rejected {task['id']}: {reason}")
                    continue
                curated.append(skill)
                self.ledger.append(
                    "candidate_proposed",
                    {"skill_id": skill["id"], "task": task["id"]},
                )
                print(f"  proposed {skill['id']} from {task['id']}")
        _save(self.state_dir, "candidates_proposed.json", curated)
        return curated

    def phase_validate(
        self, acquire_results: list[dict[str, Any]], candidates: list[dict[str, Any]]
    ) -> str:
        _, val_ids, _ = self.default_splits()
        val_ids = _ids(self.args.val_ids, val_ids)
        val_tasks = self.cfg.tasks_by_ids(val_ids)
        if not candidates:
            self.ledger.append("no_candidates", {"validation_tasks": val_ids})
            print("[validate] no candidates; keeping base policy")
            return "base"
        print(f"[validate] {len(val_tasks)} validation tasks, 3 arms")
        base_results = run_arm(self.client, val_tasks, self.cfg, policy="base")
        placebo_results = run_arm(
            self.client, val_tasks, self.cfg, policy="placebo", skills=candidates
        )
        skills_results = run_arm(
            self.client, val_tasks, self.cfg, policy="skills", skills=candidates
        )
        _save(
            self.state_dir,
            "validation_results.json",
            {
                "base": base_results,
                "placebo": placebo_results,
                "skills": skills_results,
            },
        )
        self.ledger.append(
            "validation_run",
            {
                "tasks": val_ids,
                "base_resolved": [r["task_id"] for r in base_results if r.get("resolved")],
                "placebo_resolved": [r["task_id"] for r in placebo_results if r.get("resolved")],
                "skills_resolved": [r["task_id"] for r in skills_results if r.get("resolved")],
            },
        )
        acq_ids = [r["task_id"] for r in acquire_results]
        regression_tasks = self.cfg.tasks_by_ids(acq_ids)
        print(f"[validate] regression arm: {len(regression_tasks)} acquisition tasks")
        regression_candidate = run_arm(
            self.client, regression_tasks, self.cfg, policy="skills", skills=candidates
        )
        decision = decide_promotion(
            base_results,
            skills_results,
            acquire_results,
            regression_candidate,
            self.cfg.data,
        )
        self.ledger.append(
            "promotion_decision",
            {"decision": decision, "candidate_ids": [s["id"] for s in candidates]},
        )
        print(f"[validate] decision: {'PROMOTE' if decision['promote'] else 'REJECT'}")
        print(f"  stats: {json.dumps(decision['stats'], ensure_ascii=False)}")
        print(f"  reason: {decision['reason']}")
        if decision["promote"]:
            library = SkillLibrary(self.state_dir / "skills_library.json")
            for skill in candidates:
                skill["status"] = "promoted"
                library.add(skill)
            _save(self.state_dir, "candidates_proposed.json", candidates)
            return "skills"
        proposed = SkillLibrary(self.state_dir / "candidates_proposed.json")
        for skill in candidates:
            proposed.set_status(skill["id"], "rejected", decision["reason"])
        return "base"

    def phase_sealed(self, best_policy: str) -> list[dict[str, Any]]:
        _, _, sealed_ids = self.default_splits()
        sealed_ids = _ids(self.args.sealed_ids, sealed_ids)
        tasks = self.cfg.tasks_by_ids(sealed_ids)
        skills = None
        if best_policy == "skills":
            skills = SkillLibrary(self.state_dir / "skills_library.json").promoted()
        print(f"[sealed] {len(tasks)} tasks, policy={best_policy}")
        results = run_arm(
            self.client, tasks, self.cfg, policy=best_policy, skills=skills
        )
        _save(self.state_dir, "sealed_results.json", results)
        self.ledger.append(
            "sealed_run",
            {
                "tasks": sealed_ids,
                "policy": best_policy,
                "resolved": [r["task_id"] for r in results if r.get("resolved")],
            },
        )
        print(f"[sealed] resolved {sum(1 for r in results if r.get('resolved'))}/{len(results)}")
        return results

    def phase_report(
        self,
        acquire_results: list[dict[str, Any]],
        candidates: list[dict[str, Any]],
        best_policy: str,
        sealed_results: list[dict[str, Any]],
    ) -> None:
        report = {
            "fingerprint": self.cfg.fingerprint(),
            "best_policy": best_policy,
            "acquisition": {
                "n": len(acquire_results),
                "resolved": [r["task_id"] for r in acquire_results if r.get("resolved")],
            },
            "candidates": [
                {
                    "id": s["id"],
                    "category": s["trigger_category"],
                    "keywords": s["keywords"],
                    "evidence_refs": s["evidence_refs"],
                    "status": s.get("status", "proposed"),
                }
                for s in candidates
            ],
            "sealed": {
                "policy": best_policy,
                "n": len(sealed_results),
                "resolved": [r["task_id"] for r in sealed_results if r.get("resolved")],
                "results": sealed_results,
            },
            "evidence_level": (
                "SMOKE-ONLY: results describe this machine, this task bank, and "
                "this policy; they are not evidence of generalization or recursive "
                "self-improvement."
            ),
        }
        _save(self.state_dir, "cycle_report.json", report)
        self.ledger.append("cycle_report", {"best_policy": best_policy})
        print(f"\n=== CYCLE REPORT ({best_policy}) ===")
        print(
            f"acquisition: {sum(1 for r in acquire_results if r.get('resolved'))}/{len(acquire_results)}"
        )
        print(f"candidates: {len(candidates)}")
        print(f"sealed: {sum(1 for r in sealed_results if r.get('resolved'))}/{len(sealed_results)}")
        print(f"report: {self.state_dir / 'cycle_report.json'}")

    def run(self) -> None:
        self.start()
        if self.args.phase in ("all", "validate-bank"):
            self.phase_validate_bank()
        if self.args.phase in ("all", "acquire", "diagnose", "validate", "sealed", "report"):
            acquire_results = _load(self.state_dir, "acquire_results.json")
            if acquire_results is None or self.args.phase == "acquire":
                acquire_results = self.phase_acquire()
        if self.args.phase in ("all", "diagnose", "validate", "report"):
            candidates = _load(self.state_dir, "candidates_proposed.json") or []
            if self.args.phase in ("all", "diagnose") or not candidates:
                candidates = self.phase_diagnose(acquire_results)
        if self.args.phase in ("all", "validate", "sealed", "report"):
            candidates = _load(self.state_dir, "candidates_proposed.json") or []
            best_policy = self.phase_validate(acquire_results, candidates)
        else:
            best_policy = "base"
        if self.args.phase in ("all", "sealed", "report"):
            best_policy = (
                best_policy
                if self.args.phase != "report"
                else ("skills" if SkillLibrary(self.state_dir / "skills_library.json").promoted() else "base")
            )
            sealed_results = self.phase_sealed(best_policy)
        else:
            sealed_results = _load(self.state_dir, "sealed_results.json") or []
        if self.args.phase in ("all", "report"):
            candidates = _load(self.state_dir, "candidates_proposed.json") or []
            self.phase_report(acquire_results, candidates, best_policy, sealed_results)


def main() -> None:
    parser = argparse.ArgumentParser(description="Minimal trusted self-evolving loop")
    parser.add_argument(
        "--phase",
        default="all",
        choices=["all", "validate-bank", "acquire", "diagnose", "validate", "sealed", "report"],
    )
    parser.add_argument("--smoke", action="store_true", help="small task subset")
    parser.add_argument("--acq-ids", default="")
    parser.add_argument("--val-ids", default="")
    parser.add_argument("--sealed-ids", default="")
    parser.add_argument("--endpoint", default=None)
    parser.add_argument("--model", default=None)
    parser.add_argument("--max-tokens", type=int, default=None)
    parser.add_argument("--diagnose-tokens", type=int, default=640)
    parser.add_argument(
        "--state-dir",
        default=str(ROOT / "state"),
        help="state directory (ledger, skills, results)",
    )
    args = parser.parse_args()
    Cycle(args).run()


if __name__ == "__main__":
    main()
