import sqlite3, os, datetime, glob, re

home = r"C:\Users\Administrator\AppData\Roaming\haolo_desktop\thread-groups\default\agent\haolo-ai-home"
db = os.path.join(home, "state_5.sqlite")
con = sqlite3.connect("file:"+db.replace("\\","/")+"?mode=ro", uri=True)
con.row_factory = sqlite3.Row
cur = con.cursor()

kws = ["审稿", "review", "four_papers", "四篇", "审阅", "评审", "revision", "round", "R7", "R8", "R9", "R10", "R11", "R12", "paper1", "paper2", "paper3", "paper4", "paper5"]

rows = cur.execute("SELECT * FROM threads ORDER BY created_at").fetchall()
print("agent threads total:", len(rows))

for r in rows:
    d = dict(r)
    blob = " ".join(str(d.get(k) or "") for k in ("title", "first_user_message", "preview"))
    low = blob.lower()
    hits = [k for k in ["审稿", "审阅", "评审", "review", "four_papers", "四篇", "revision", "round"] if k.lower() in low]
    rp = d.get("rollout_path") or ""
    exists = os.path.exists(rp) if rp else False
    size = os.path.getsize(rp) if exists else 0
    ca = datetime.datetime.fromtimestamp(d["created_at"]).strftime("%m-%d %H:%M")
    ua = datetime.datetime.fromtimestamp(d["updated_at"]).strftime("%m-%d %H:%M") if d["updated_at"] else ""
    if hits:
        sn = (d.get("first_user_message") or "")
        if "<haolo_windows_utf8_reminder>" in sn:
            sn = sn.split("</haolo_windows_utf8_reminder>")[-1]
        if "<haolo_media_routing_reminder>" in sn:
            sn = sn.split("</haolo_media_routing_reminder>")[-1]
        sn = " ".join(sn.split())[:80]
        print(f"[{ca} -> {ua}] arch={d['archived']} sub={d.get('thread_source')} id={d['id'][:13]} size={size/1024:.0f}KB hits={hits} | {sn}")

# also scan all rollout files for review-round markers
print()
print("=== rollout files containing 'REVIEW_FOUR_PAPERS' / 'four_papers_review' / '审稿' ===")
files = glob.glob(os.path.join(home, "sessions", "**", "rollout-*.jsonl"), recursive=True)
files += glob.glob(os.path.join(home, "archived_sessions", "rollout-*.jsonl"))
for fp in files:
    with open(fp, "rb") as f:
        data = f.read()
    markers = []
    for kw in [b"REVIEW_FOUR_PAPERS", b"four_papers_review", b"审稿", b"Review-Agent", b"Round 21", b"REVISION_R", b"ACCEPT_PLAN"]:
        if kw in data:
            markers.append(kw.decode("utf-8", "ignore"))
    if markers:
        mt = datetime.datetime.fromtimestamp(os.path.getmtime(fp)).strftime("%m-%d %H:%M")
        print(f"{mt} | {os.path.basename(fp)} | {os.path.getsize(fp)/1024:.0f}KB | markers={markers}")
con.close()
