import struct
import sys

path = sys.argv[1]

with open(path, "rb") as f:
    data = f.read()

off = 4 + 4 + 8 + 8

def read_string(d, o):
    n = struct.unpack_from("<Q", d, o)[0]
    o += 8
    return d[o:o+n].decode("utf-8", errors="replace"), o + n

def read_value(d, o, vtype):
    if vtype == 8:
        return read_string(d, o)
    if vtype == 9:
        atype = struct.unpack_from("<I", d, o)[0]
        o += 4
        n = struct.unpack_from("<Q", d, o)[0]
        o += 8
        items = []
        for _ in range(n):
            v, o = read_value(d, o, atype)
            items.append(v)
        return items, o
    if vtype == 7:
        return struct.unpack_from("<?", d, o)[0], o + 1
    if vtype == 6:
        return struct.unpack_from("<f", d, o)[0], o + 4
    if vtype == 12:
        return struct.unpack_from("<d", d, o)[0], o + 8
    if vtype in (4, 5):
        return struct.unpack_from("<i", d, o)[0], o + 4
    if vtype in (10, 11):
        return struct.unpack_from("<q", d, o)[0], o + 8
    if vtype in (0, 1, 2, 3):
        return struct.unpack_from("<H", d, o)[0], o + 2
    raise ValueError(f"unsupported type {vtype}")

kv_count = struct.unpack_from("<Q", data, off - 8)[0]
special = {}
for _ in range(kv_count):
    key, off = read_string(data, off)
    vtype = struct.unpack_from("<I", data, off)[0]
    off += 4
    val, off = read_value(data, off, vtype)
    special[key] = val

tokens = special.get("tokenizer.ggml.tokens", [])
for key, idx_key in (
    ("bos", "tokenizer.ggml.bos_token_id"),
    ("eos", "tokenizer.ggml.eos_token_id"),
    ("pad", "tokenizer.ggml.pad_token_id"),
    ("unk", "tokenizer.ggml.unknown_token_id"),
):
    idx = special.get(idx_key)
    if idx is not None and idx < len(tokens):
        print(f"{key}_token_id={idx} text={tokens[idx]!r}")
    else:
        print(f"{key}_token_id={idx} (no text)")
