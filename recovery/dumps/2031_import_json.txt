import json
import struct
import sys

path = sys.argv[1]

with open(path, "rb") as f:
    data = f.read()

off = 0
magic = data[off:off+4]
off += 4
version = struct.unpack_from("<I", data, off)[0]
off += 4
tensor_count = struct.unpack_from("<Q", data, off)[0]
off += 8
kv_count = struct.unpack_from("<Q", data, off)[0]
off += 8

print(f"magic={magic.decode('ascii')} version={version} tensors={tensor_count} kv={kv_count}")

TYPE_NAMES = {
    0: "uint8", 1: "int8", 2: "uint16", 3: "int16", 4: "uint32", 5: "int32",
    6: "float32", 7: "bool", 8: "string", 9: "array", 10: "uint64", 11: "int64",
    12: "float64",
}

def read_string(d, o):
    n = struct.unpack_from("<Q", d, o)[0]
    o += 8
    s = d[o:o+n].decode("utf-8", errors="replace")
    return s, o + n

def read_value(d, o, vtype):
    if vtype == 8:
        return read_string(d, o)
    if vtype == 7:
        return struct.unpack_from("<?", d, o)[0], o + 1
    if vtype == 6:
        return struct.unpack_from("<f", d, o)[0], o + 4
    if vtype == 12:
        return struct.unpack_from("<d", d, o)[0], o + 8
    if vtype in (4, 5):
        return struct.unpack_from("<i", d, o)[0], o + 4
    if vtype in (10, 11):
        return struct.unpack_from("<q", d, o)[0], o + 8
    if vtype in (0, 1, 2, 3):
        return struct.unpack_from("<H", d, o)[0], o + 2
    if vtype == 9:
        atype = struct.unpack_from("<I", d, o)[0]
        o += 4
        n = struct.unpack_from("<Q", d, o)[0]
        o += 8
        items = []
        for _ in range(n):
            v, o = read_value(d, o, atype)
            items.append(v)
        return items, o
    raise ValueError(f"unsupported type {vtype}")

for i in range(kv_count):
    key, off = read_string(data, off)
    vtype = struct.unpack_from("<I", data, off)[0]
    off += 4
    val, off = read_value(data, off, vtype)
    tname = TYPE_NAMES.get(vtype, str(vtype))
    if isinstance(val, str):
        print(f"{key} ({tname}): {val}")
    else:
        print(f"{key} ({tname}): {json.dumps(val, ensure_ascii=False)}")
