// user_perspective_e2e.mjs —— 用户视角全动线测试（她 + 他）
// 不测接口，只走真实 UI：首开空态 → 今日一句话 → 树洞 → 打卡 → 进度空态 →
// 题库看题 → 试卷模式交卷 → 记自测 → 悄悄话；督学端全程旁观验证陪伴链路。
import puppeteer from 'puppeteer-core'
import { mkdirSync, writeFileSync } from 'node:fs'

const URL = 'http://127.0.0.1:4173'
const EDGE = 'C:\\Program Files (x86)\\Microsoft\\Edge\\Application\\msedge.exe'
const SHOTS = 'C:/Users/Administrator/AppData/Roaming/haolo_desktop/thread-groups/default/科研/outputs/user-e2e'
mkdirSync(SHOTS, { recursive: true })

const ts = Date.now()
const EMAIL_STU = `u-stu-${ts}@example.com`
const EMAIL_SUP = `u-sup-${ts}@example.com`
const PASS = 'Test123456'
const WORD = '今天数学卡住了，但没放弃'
const TREE = '好累，但还能再学半小时'
const MSG = `用户视角消息-${ts}`
const REPLY = `用户视角回复-${ts}`

const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
const results = []
const pageErrors = []
const shots = []
const ok = (name, pass, extra = '') => {
  results.push({ name, pass: !!pass, extra })
  console.log((pass ? 'PASS' : 'FAIL') + ' ' + name + (extra ? ' :: ' + extra : ''))
}

const browser = await puppeteer.launch({
  executablePath: EDGE,
  headless: 'new',
  args: ['--no-sandbox', '--disable-gpu', '--disable-dev-shm-usage']
})

async function newPage(ctx) {
  const p = await ctx.newPage()
  await p.setViewport({ width: 412, height: 915, isMobile: true, hasTouch: true })
  p.on('pageerror', (e) => pageErrors.push('[pageerror] ' + e.message))
  p.on('console', (m) => { if (m.type() === 'error') pageErrors.push('[console] ' + m.text().slice(0, 300)) })
  return p
}

async function clickByText(page, text, { partial = false } = {}) {
  const hit = await page.evaluate(([t, p]) => {
    const btns = Array.from(document.querySelectorAll('button'))
    const b = btns.find((x) => p ? x.textContent.includes(t) : x.textContent.trim() === t)
    if (b) { b.click(); return true }
    return false
  }, [text, partial])
  if (!hit) throw new Error('button not found: ' + text)
}

async function typeByPlaceholder(page, ph, val) {
  const hit = await page.evaluate(([p, v]) => {
    const els = Array.from(document.querySelectorAll('input, textarea'))
    const el = els.find((x) => (x.placeholder || '').startsWith(p))
    if (!el) return false
    const proto = el.tagName === 'TEXTAREA' ? window.HTMLTextAreaElement.prototype : window.HTMLInputElement.prototype
    Object.getOwnPropertyDescriptor(proto, 'value').set.call(el, v)
    el.dispatchEvent(new Event('input', { bubbles: true }))
    return true
  }, [ph, val])
  if (!hit) throw new Error('input not found: ' + ph)
}

async function waitText(page, t, timeout = 60000) {
  await page.waitForFunction((x) => document.body.innerText.includes(x), { timeout }, t)
}

async function waitGone(page, t, timeout = 60000) {
  await page.waitForFunction((x) => !document.body.innerText.includes(x), { timeout }, t)
}

async function shot(page, name) {
  const f = SHOTS + '/' + name + '.png'
  await page.screenshot({ path: f })
  shots.push(f)
}

try {
  const ctxA = await browser.createBrowserContext()
  const ctxB = await browser.createBrowserContext()
  const she = await newPage(ctxA)
  const he = await newPage(ctxB)

  // ============ 她：首开 ============
  await she.goto(URL, { waitUntil: 'networkidle0', timeout: 60000 })
  await clickByText(she, '注册')
  await clickByText(she, '学员（她）')
  await typeByPlaceholder(she, '起个名字', '测试学员')
  await typeByPlaceholder(she, 'you@example.com', EMAIL_STU)
  await typeByPlaceholder(she, '至少 6 位', PASS)
  await clickByText(she, '创建账号')
  await waitText(she, '邀请码', 60000)
  await shot(she, '01-她-首开今日页')
  ok('她首开进入今日页', true)
  await waitText(she, '今天还没打卡')
  await waitText(she, '今日一句话')
  await waitText(she, '树洞')
  ok('首开空态：未打卡 / 今日一句话 / 树洞 都在', true)

  // 取邀请码
  await clickByText(she, '邀请码')
  await she.waitForSelector('.code-display', { timeout: 10000 })
  const code = (await she.$eval('.code-display', (el) => el.textContent.trim())).toUpperCase()
  await clickByText(she, '关闭')
  ok('她拿到 6 位邀请码', /^[A-Z2-9]{6}$/.test(code), code)

  // ============ 他：注册配对 + 空态旁观 ============
  await he.goto(URL, { waitUntil: 'networkidle0', timeout: 60000 })
  await clickByText(he, '注册')
  await clickByText(he, '督学（你）')
  await typeByPlaceholder(he, '向学员索取 6 位邀请码', code)
  await typeByPlaceholder(he, '起个名字', '测试督学')
  await typeByPlaceholder(he, 'you@example.com', EMAIL_SUP)
  await typeByPlaceholder(he, '至少 6 位', PASS)
  await clickByText(he, '创建账号')
  await waitText(he, '远程陪伴', 60000)
  await waitText(he, '她今天还没打卡', 60000)
  await shot(he, '02-他-空态看到没打卡')
  ok('他配对成功，看到「她今天还没打卡」', true)

  // ============ 她：写今日一句话 + 树洞 + 打卡 ============
  await typeByPlaceholder(she, '比如：今天数学卡住了', WORD)
  await waitText(she, '100')
  await shot(she, '03-她-写下今日一句话')
  ok('她写下今日一句话（自动保存）', true)

  await she.evaluate(() => {
    const els = Array.from(document.querySelectorAll('*'))
    const el = els.find((x) => x.textContent.trim() === '树洞')
    const card = el && el.closest('.card')
    if (card) card.click()
  })
  await waitText(she, '只有你能看到')
  await typeByPlaceholder(she, '烦了、累了、想吐槽', TREE)
  await shot(she, '04-她-树洞倾诉')
  await clickByText(she, '收起')
  ok('树洞可写可收，隐私文案在', true)

  await typeByPlaceholder(she, '今天想对自己说的一句话', '今天也要加油一点点')
  await clickByText(she, '一键打卡')
  await waitText(she, '已打卡', 60000)
  await shot(she, '05-她-打卡成功')
  ok('她一键打卡成功', true)

  // ============ 她：进度页空态 ============
  await clickByText(she, '进度')
  await waitText(she, '还没有真题自测记录', 60000)
  await waitText(she, '还没用真题自测过')
  await shot(she, '06-她-进度页空态')
  ok('进度页空态：雷达/离岸线如实显示「未测」', true)

  // ============ 她：题库看题 ============
  await clickByText(she, '备考')
  await waitText(she, '展开 408 题库')
  await clickByText(she, '展开 408 题库')
  await waitText(she, '真题库', 60000)
  await clickByText(she, '真题库')
  await waitText(she, '真题库共', 60000)
  await shot(she, '07-她-真题库加载')
  await she.waitForFunction(() => {
    const btns = Array.from(document.querySelectorAll('button'))
    return btns.some((b) => b.textContent.includes('显示答案'))
  }, { timeout: 30000 })
  await clickByText(she, '显示答案', { partial: true })
  await waitText(she, '答案：', 15000)
  await shot(she, '08-她-看答案解析')
  ok('真题库可看题、可揭答案', true)

  // ============ 她：试卷模式交卷 ============
  await clickByText(she, '试卷模式')
  await she.waitForSelector('select', { timeout: 15000 })
  await she.evaluate(() => {
    const sel = document.querySelector('select')
    const opts = Array.from(sel.options).map((o) => o.value).filter(Boolean)
    if (opts.length === 0) return
    Object.getOwnPropertyDescriptor(window.HTMLSelectElement.prototype, 'value').set.call(sel, opts[0])
    sel.dispatchEvent(new Event('change', { bubbles: true }))
  })
  await she.waitForFunction(() => {
    const btns = Array.from(document.querySelectorAll('button'))
    return btns.some((b) => b.textContent.trim().startsWith('A.'))
  }, { timeout: 30000 })
  await she.evaluate(() => {
    const btns = Array.from(document.querySelectorAll('button'))
    const b = btns.find((x) => x.textContent.trim().startsWith('A.'))
    if (b) b.click()
  })
  await clickByText(she, '交卷')
  await waitText(she, '得分', 30000)
  await shot(she, '09-她-试卷交卷得分')
  ok('试卷模式：选年份 → 作答 → 交卷出分', true)

  // ============ 她：记一次自测 ============
  await clickByText(she, '返回题库')
  await she.evaluate(() => window.scrollTo(0, 0))
  await waitText(she, '记一次自测')
  await typeByPlaceholder(she, '总题数', '10')
  await typeByPlaceholder(she, '做对', '8')
  await clickByText(she, '记录这次自测')
  await waitText(she, '80%', 30000)
  await shot(she, '10-她-记录自测')
  ok('她手动记录自测 10 对 8 → 80%', true)

  // ============ 她：进度页看到数据 ============
  await clickByText(she, '进度')
  await waitText(she, '80%', 60000)
  await shot(she, '11-她-进度页有数据')
  ok('进度页/离岸线用真题口径显示 80%', true)

  // ============ 她：发悄悄话 ============
  await clickByText(she, '悄悄话')
  await she.waitForFunction(() => !!document.querySelector('input[placeholder^="对 "]'), { timeout: 15000 })
  await typeByPlaceholder(she, '对 ', MSG)
  await clickByText(she, '发送')
  await waitText(she, MSG, 30000)
  await shot(she, '12-她-发出悄悄话')
  ok('她发出悄悄话', true)

  // ============ 他：看到她写的话 + 数据同步 ============
  await waitText(he, WORD, 60000)
  await shot(he, '13-他-看到她写的今日一句话')
  ok('他看到她写的今日一句话（陪伴回显）', true)

  await waitGone(he, '她今天还没打卡', 60000)
  ok('他端实时看到她已打卡（复活卡提示消失）', true)

  await clickByText(he, '展开')
  await waitText(he, '真题平均正确率', 15000)
  await waitText(he, '真题 2 次', 60000)
  await shot(he, '14-他-看到她的自测数据')
  ok('他看到她 2 次真题记录（10 对 8 + 试卷 1 题）', true)

  // ============ 他：收到并回复 ============
  await clickByText(he, '悄悄话')
  await waitText(he, MSG, 60000)
  await shot(he, '15-他-收到她的悄悄话')
  ok('他收到她的悄悄话', true)
  await he.waitForFunction(() => !!document.querySelector('input[placeholder^="对 "]'), { timeout: 15000 })
  await typeByPlaceholder(he, '对 ', REPLY)
  await clickByText(he, '发送')
  await waitText(he, REPLY, 30000)
  await clickByText(she, '悄悄话')
  await waitText(she, REPLY, 60000)
  await shot(she, '16-她-收到他的回复')
  ok('双向悄悄话闭环', true)
} catch (e) {
  ok('执行中断：' + e.message, false)
} finally {
  await browser.close()
}

  ts: new Date().toISOString(),
  accounts: { student: EMAIL_STU, supervisor: EMAIL_SUP, password: PASS },
  results,
  pageErrors: errs,
  screenshots: shots
}

const passCount = results.filter((r) => r.pass).length
console.log(`\n结果: ${passCount}/${results.length} 通过`)
process.exit(results.every((r) => r.pass) ? 0 : 1)
