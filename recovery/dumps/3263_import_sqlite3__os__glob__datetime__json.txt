import sqlite3, os, glob, datetime, json

base = r"C:\Users\Administrator\AppData\Roaming\haolo_desktop\thread-groups\default"
groups = [("默认分组", os.path.join(base, "haolo-ai-home")),
          ("论文", os.path.join(base, "论文", "haolo-ai-home")),
          ("科研", os.path.join(base, "科研", "haolo-ai-home")),
          ("agent", os.path.join(base, "agent", "haolo-ai-home")),
          ("app", os.path.join(base, "app", "haolo-ai-home")),
          ("归档", os.path.join(base, "归档", "haolo-ai-home"))]

for gname, home in groups:
    print("="*15, gname, "="*15)
    db = os.path.join(home, "state_5.sqlite")
    if not os.path.exists(db):
        print("no state db")
        continue
    con = sqlite3.connect("file:"+db.replace("\\","/")+"?mode=ro", uri=True)
    con.row_factory = sqlite3.Row
    cur = con.cursor()
    rows = cur.execute("SELECT * FROM threads").fetchall()
    print("threads in index:", len(rows))

    missing = []
    zero = []
    index_refs = set()
    for r in rows:
        d = dict(r)
        rp = d.get("rollout_path") or ""
        if rp:
            index_refs.add(os.path.normpath(rp))
            if not os.path.exists(rp):
                missing.append((d.get("id"), rp, d.get("created_at"), d.get("updated_at")))
            else:
                sz = os.path.getsize(rp)
                if sz == 0:
                    zero.append((d.get("id"), rp, d.get("created_at")))
    print("threads with rollout_path missing file:", len(missing))
    for m in missing[:30]:
        print("  MISSING:", m)
    print("threads with 0-byte rollout:", len(zero))
    for z in zero[:30]:
        print("  ZERO:", z)

    # recency stats
    rec_n = sum(1 for r in rows if (r["recency_at"] is None and r["recency_at_ms"] is None))
    print("threads with null recency_at:", rec_n)

    # files on disk not referenced
    on_disk = set()
    for sub in ("sessions", "archived_sessions"):
        p = os.path.join(home, sub)
        if os.path.isdir(p):
            for f in glob.glob(os.path.join(p, "**", "rollout-*.jsonl"), recursive=True):
                on_disk.add(os.path.normpath(f))
    orphan = on_disk - index_refs
    print("rollout files on disk:", len(on_disk), "not in index:", len(orphan))
    for o in list(orphan)[:30]:
        print("  ORPHAN:", o, os.path.getsize(o) if os.path.exists(o) else "-")

    # files in .tmp
    tmpp = os.path.join(home, ".tmp")
    if os.path.isdir(tmpp):
        tfiles = [f for f in glob.glob(os.path.join(tmpp, "**", "*"), recursive=True) if os.path.isfile(f)]
        print(".tmp files:", len(tfiles))
        for f in tfiles[:20]:
            print("  TMP:", f, os.path.getsize(f))
    con.close()

print()
print("=== rollout sizes histogram (all groups) ===")
for gname, home in groups:
    sizes = []
    for sub in ("sessions", "archived_sessions"):
        p = os.path.join(home, sub)
        if os.path.isdir(p):
            for f in glob.glob(os.path.join(p, "**", "rollout-*.jsonl"), recursive=True):
                sizes.append(os.path.getsize(f))
    if sizes:
        small = sum(1 for s in sizes if s < 1024)
        tiny = sum(1 for s in sizes if s < 512)
        print(gname, "files:", len(sizes), "size<1KB:", small, "size<512B:", tiny, "min:", min(sizes), "max:", max(sizes))
