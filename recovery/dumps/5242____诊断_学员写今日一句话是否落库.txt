let id = 0
const send = (m, p) => new Promise((res, rej) => {
  const i = ++id
  const to = setTimeout(() => rej(new Error('timeout ' + m)), 12000)
  const h = (ev) => {
    const d = JSON.parse(ev.data)
    if (d.id === i) { clearTimeout(to); ws.removeEventListener('message', h); d.error ? rej(new Error(d.error.message)) : res(d.result) }
  }
  ws.addEventListener('message', h)
  ws.send(JSON.stringify({ id: i, method: m, params: p }))
})
const errors = []
ws.addEventListener('message', (ev) => {
  const d = JSON.parse(ev.data)
  if (d.method === 'Runtime.exceptionThrown') errors.push('EXC: ' + (d.params.exceptionDetails.text || '') + ' ' + (d.params.exceptionDetails.exception?.description || '').slice(0, 200))
  if (d.method === 'Runtime.consoleAPICalled' && d.params.type === 'error') errors.push('CONSOLE: ' + d.params.args.map((a) => a.value || a.description || '').join(' ').slice(0, 200))
})
