import sqlite3, os, datetime

base = r"C:\Users\Administrator\AppData\Roaming\haolo_desktop\thread-groups\default"
groups = [("默认分组", os.path.join(base, "haolo-ai-home")),
          ("论文", os.path.join(base, "论文", "haolo-ai-home")),
          ("科研", os.path.join(base, "科研", "haolo-ai-home")),
          ("agent", os.path.join(base, "agent", "haolo-ai-home")),
          ("app", os.path.join(base, "app", "haolo-ai-home")),
          ("归档", os.path.join(base, "归档", "haolo-ai-home"))]

for gname, home in groups:
    db = os.path.join(home, "state_5.sqlite")
    if not os.path.exists(db):
        continue
    con = sqlite3.connect("file:"+db.replace("\\","/")+"?mode=ro", uri=True)
    con.row_factory = sqlite3.Row
    cur = con.cursor()
    total = cur.execute("SELECT COUNT(*) FROM threads").fetchone()[0]
    user_ev = cur.execute("SELECT COUNT(*) FROM threads WHERE has_user_event=1").fetchone()[0]
    archived = cur.execute("SELECT COUNT(*) FROM threads WHERE archived=1").fetchone()[0]
    print("---", gname, "total:", total, "| has_user_event=1:", user_ev, "| archived:", archived)

    # distinct values of interesting columns
    for col in ("thread_source", "history_mode", "source", "agent_path"):
        vals = cur.execute(f"SELECT {col}, COUNT(*) FROM threads GROUP BY {col} ORDER BY 2 DESC LIMIT 8").fetchall()
        if vals and vals[0][0] is not None:
            print("   ", col, ":", {str(v[0])[:40]: v[1] for v in vals})

    # how many have a first_user_message (i.e., real user-started)
    fum = cur.execute("SELECT COUNT(*) FROM threads WHERE first_user_message IS NOT NULL AND length(first_user_message)>0").fetchone()[0]
    print("    first_user_message set:", fum)

    # distribution of user_ev by month (created_at)
    rows = cur.execute("SELECT created_at, has_user_event, archived FROM threads").fetchall()
    by_month = {}
    for r in rows:
        m = datetime.datetime.fromtimestamp(r[0]).strftime("%Y-%m")
        by_month.setdefault(m, [0, 0])
        by_month[m][0] += 1
        if r[1] == 1 and r[2] == 0:
            by_month[m][1] += 1
    print("    by month (total, user-event non-archived):", dict(by_month))
    con.close()
